// Stars
bot.action(/^pay_stars_(.+)$/, async (ctx) => {
  await ctx.answerCbQuery();
  try { await payments.sendStarsInvoice(ctx, ctx.match[1]); await ctx.deleteMessage().catch(() => {}); }
  catch (e) { await ctx.reply(`❌ Ошибка: ${e.message}`); }
});

// ЮKassa
bot.action(/^pay_yukassa_(.+)$/, async (ctx) => {
  await ctx.answerCbQuery();
  const productId = ctx.match[1];
  if (!process.env.YUKASSA_TOKEN) {
    const product = payments.getProducts()[productId];
    return ctx.reply(
      `💳 *Оплата картой через ЮKassa*\n\n` +
      `📦 ${product?.name} — *${product?.price_rub} ₽*\n\n` +
      `⚠️ ЮKassa пока не подключена.\nНапиши @Imalisa_boychuk — она примет оплату вручную и активирует доступ!`,
      { parse_mode: 'Markdown', ...Markup.inlineKeyboard([[Markup.button.callback('🔙 Назад', 'buy_premium')]]) }
    );
  }
  try { await payments.sendYukassaInvoice(ctx, productId); await ctx.deleteMessage().catch(() => {}); }
  catch (e) { await ctx.reply(`❌ Ошибка: ${e.message}`); }
});

// ЮMoney — генерируем ссылку
bot.action(/^pay_ymoney_(.+)$/, async (ctx) => {
  await ctx.answerCbQuery();
  const productId = ctx.match[1];
  const product = payments.getProducts()[productId];

  if (!process.env.YMONEY_WALLET) {
    return ctx.reply(
      `💛 *Оплата через ЮMoney*\n\n` +
      `📦 ${product?.name} — *${product?.price_rub} ₽*\n\n` +
      `⚠️ ЮMoney пока не настроена.\nНапиши @Imalisa_boychuk — она пришлёт реквизиты!`,
      { parse_mode: 'Markdown', ...Markup.inlineKeyboard([[Markup.button.callback('🔙 Назад', 'buy_premium')]]) }
    );
  }

  const link = payments.generateYmoneyLink(productId, ctx.from.id);
  if (!link) return ctx.reply('❌ Ошибка генерации ссылки');

  // Сохраняем ожидающий платёж
  await payments.savePendingYmoney(link.label, {
    userId: ctx.from.id,
    productId,
    amount: link.amount,
  });

  await ctx.editMessageText(
    `💛 *Оплата через ЮMoney*\n\n` +
    `📦 *${product?.name}*\n` +
    `💰 Сумма: *${link.amount} ₽*\n\n` +
    `1️⃣ Нажми кнопку ниже — откроется форма оплаты\n` +
    `2️⃣ Оплати картой или с кошелька ЮMoney\n` +
    `3️⃣ После оплаты нажми *«✅ Я оплатил»*\n\n` +
    `_Доступ активируется автоматически!_`,
    {
      parse_mode: 'Markdown',
      ...Markup.inlineKeyboard([
        [Markup.button.url('💛 Оплатить через ЮMoney', link.url)],
        [Markup.button.callback(`✅ Я оплатил — ${link.label.slice(-8)}`, `ymoney_confirm_${link.label}`)],
        [Markup.button.callback('🔙 Назад', 'buy_premium')],
      ])
    }
  );
});

// Подтверждение оплаты ЮMoney пользователем
bot.action(/^ymoney_confirm_(.+)$/, async (ctx) => {
  await ctx.answerCbQuery('⏳ Проверяю оплату...');
  const label = ctx.match[1];

  // Отправляем заявку на ручную проверку админу
  for (const adminId of ADMIN_IDS) {
    try {
      await ctx.telegram.sendMessage(adminId,
        `💛 *Заявка на проверку ЮMoney*\n\n` +
        `👤 ${ctx.from.first_name} (@${ctx.from.username || ctx.from.id})\n` +
        `🏷 Label: \`${label}\`\n\n` +
        `Проверь в личном кабинете ЮMoney и подтверди:`,
        {
          parse_mode: 'Markdown',
          reply_markup: {
            inline_keyboard: [[
              { text: '✅ Подтвердить оплату', callback_data: `admin_ymoney_ok_${label}` },
              { text: '❌ Отклонить', callback_data: `admin_ymoney_no_${label}` },
            ]]
          }
        }
      );
    } catch {}
  }

  await ctx.editMessageText(
    `⏳ *Заявка отправлена на проверку!*\n\n` +
    `Администратор проверит оплату в течение нескольких часов и активирует доступ.\n\n` +
    `Если есть вопросы — @Imalisa_boychuk`,
    { parse_mode: 'Markdown', ...Markup.inlineKeyboard([[Markup.button.callback('🏠 Главное меню', 'main_menu')]]) }
  );
});

// Админ подтверждает ЮMoney
bot.action(/^admin_ymoney_ok_(.+)$/, async (ctx) => {
  if (!ADMIN_IDS.includes(ctx.from.id)) return ctx.answerCbQuery('⛔');
  await ctx.answerCbQuery('✅ Активирую...');
  const label = ctx.match[1];
  const result = await payments.confirmYmoney(label);
  if (result) {
    await ctx.editMessageText(`✅ Premium активирован для пользователя ${result.userId}!`);
    try {
      const product = payments.getProducts()[result.productId];
      await ctx.telegram.sendMessage(result.userId,
        `🎉 *Оплата подтверждена!*\n\n📦 ${product?.name}\n✅ Premium активирован!\nДо: *${new Date(result.premiumData.expires_at).toLocaleDateString('ru-RU')}*`,
        { parse_mode: 'Markdown', ...Markup.inlineKeyboard([[{ text: '🏠 Главное меню', callback_data: 'main_menu' }]]) }
      );
    } catch {}
  } else {
    await ctx.editMessageText('❌ Платёж не найден или уже обработан.');
  }
});

bot.action(/^admin_ymoney_no_(.+)$/, async (ctx) => {
  if (!ADMIN_IDS.includes(ctx.from.id)) return ctx.answerCbQuery('⛔');
  await ctx.answerCbQuery('❌');
  await ctx.editMessageText('❌ Платёж отклонён.');
});
